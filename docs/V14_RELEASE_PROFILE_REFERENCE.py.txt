from imperion.config import VERSION, load_settings


def test_release_version_and_conservative_default_profile():
    s = load_settings({})
    assert VERSION == "14.3.0"
    assert s.mode == "dry"
    assert s.risk_per_trade_pct == 0.25
    assert s.max_concurrent_positions == 1
    assert s.max_open_risk_pct == 0.50
    assert s.max_daily_new_risk_pct == 1.0
    assert s.daily_loss_limit_pct == 1.5
    assert s.max_drawdown_pct == 5.0
    assert s.leverage == 2 and s.max_leverage == 2
    assert not s.live_confirm
